  <!-- SCRIPT CON AUDIO SYNTH, CANVAS ECG Y LÓGICA CLÍNICA -->
  <script>
    // ==========================================
    // 1. SINTETIZADOR DE AUDIO (Web Audio API)
    // ==========================================
    let audioCtx = null;
    let isSoundOn = true;

    function getAudioCtx() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
      if (audioCtx.state === 'suspended') {
        audioCtx.resume();
      }
      return audioCtx;
    }

    // Toggle de sonido
    const soundToggle = document.getElementById('soundToggle');
    const soundIcon = document.getElementById('soundIcon');
    const soundLabel = document.getElementById('soundLabel');

    soundToggle.addEventListener('click', () => {
      isSoundOn = !isSoundOn;
      soundIcon.textContent = isSoundOn ? '🔊' : '🔇';
      soundLabel.textContent = isSoundOn ? 'Monitor: ACTIVO' : 'Monitor: MUTED';
    });

    // Beep médico del monitor cardíaco
    function playBeep(pitch = 750, duration = 0.08) {
      if (!isSoundOn) return;
      try {
        const ctx = getAudioCtx();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(pitch, ctx.currentTime);
        gain.gain.setValueAtTime(0.12, ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start();
        osc.stop(ctx.currentTime + duration + 0.01);
      } catch (e) {}
    }

    // Carga eléctrica y ZAP de Desfibrilador
    function playDefibZap() {
      if (!isSoundOn) return;
      try {
        const ctx = getAudioCtx();
        
        // 1. Tono de carga ascendente
        const oscCharge = ctx.createOscillator();
        const gainCharge = ctx.createGain();
        oscCharge.type = 'sawtooth';
        oscCharge.frequency.setValueAtTime(220, ctx.currentTime);
        oscCharge.frequency.exponentialRampToValueAtTime(1400, ctx.currentTime + 0.8);
        gainCharge.gain.setValueAtTime(0.01, ctx.currentTime);
        gainCharge.gain.linearRampToValueAtTime(0.18, ctx.currentTime + 0.7);
        gainCharge.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.85);
        oscCharge.connect(gainCharge);
        gainCharge.connect(ctx.destination);
        oscCharge.start();
        oscCharge.stop(ctx.currentTime + 0.85);

        // 2. Descarga / ZAP estruendoso
        setTimeout(() => {
          const oscZap = ctx.createOscillator();
          const gainZap = ctx.createGain();
          oscZap.type = 'square';
          oscZap.frequency.setValueAtTime(120, ctx.currentTime);
          oscZap.frequency.exponentialRampToValueAtTime(40, ctx.currentTime + 0.35);

          gainZap.gain.setValueAtTime(0.4, ctx.currentTime);
          gainZap.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.35);

          oscZap.connect(gainZap);
          gainZap.connect(ctx.destination);
          oscZap.start();
          oscZap.stop(ctx.currentTime + 0.36);
        }, 800);
      } catch (e) {}
    }

    // Clic sutil de pastilla / cápsula
    function playPillSound() {
      if (!isSoundOn) return;
      try {
        const ctx = getAudioCtx();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(800, ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(400, ctx.currentTime + 0.06);
        gain.gain.setValueAtTime(0.2, ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.06);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start();
        osc.stop(ctx.currentTime + 0.07);
      } catch (e) {}
    }

    // 2. MONITOR ECG EN CANVAS (DINÁMICO)
    // ==========================================
    const canvas = document.getElementById('ecgCanvas');
    const ctx = canvas.getContext('2d');
    const bpmDisplay = document.getElementById('bpmDisplay');
    const batteryDisplay = document.getElementById('batteryDisplay');

    let isRevived = false;
    let animFrameId = null;

    // Buffer de puntos para el barrido del osciloscopio
    let width = 0;
    let height = 0;
    let scanX = 0;
    let lastY = 0;

    function resizeCanvas() {
      const rect = canvas.getBoundingClientRect();
      width = canvas.width = rect.width;
      height = canvas.height = rect.height;
      lastY = height / 2;
      ctx.fillStyle = '#051811';
      ctx.fillRect(0, 0, width, height);
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    let beatCounter = 0;

    function drawECG() {
      // Borrar una franja delante del haz de luz (efecto fósforo de monitor)
      const clearWidth = 14;
      ctx.fillStyle = '#051811';
      ctx.fillRect(scanX, 0, clearWidth, height);

      let currentY = height / 2;
      beatCounter++;

      // ECG lento o reactivado
      const cycleLength = isRevived ? 40 : 110;
      const progress = beatCounter % cycleLength;

      if (!isRevived) {
        // Trazado lento y débil
        if (progress === 10) {
          currentY = height / 2 - 12; // Pequeña onda P
        } else if (progress === 15) {
          currentY = height / 2 + 8; // Onda Q
        } else if (progress === 18) {
          currentY = height / 2 - 35; // Onda R baja
          playBeep(480, 0.09); // Beep lento y grave
        } else if (progress === 21) {
          currentY = height / 2 + 12; // Onda S
        } else if (progress === 30) {
          currentY = height / 2 - 10; // Onda T tímida
        } else {
          // Línea casi plana con leve ruido
          currentY = height / 2 + (Math.random() * 2 - 1);
        }
      } else {
        // Trazado enérgico, juvenil y vivaz
        if (progress === 6) {
          currentY = height / 2 - 18; // Onda P sana
        } else if (progress === 9) {
          currentY = height / 2 + 14; // Q
        } else if (progress === 12) {
          currentY = height / 2 - 65; // Onda R alta y fuerte
          playBeep(880, 0.07); // Beep alegre y claro
        } else if (progress === 15) {
          currentY = height / 2 + 25; // S
        } else if (progress === 22) {
          currentY = height / 2 - 24; // Onda T plena
        } else {
          currentY = height / 2 + (Math.random() * 2 - 1);
        }
      }

      // Dibujar línea del ECG
      ctx.beginPath();
      ctx.moveTo(scanX - 2 < 0 ? 0 : scanX - 2, lastY);
      ctx.lineTo(scanX, currentY);
      ctx.lineWidth = 2.4;
      ctx.strokeStyle = isRevived ? '#2DF89F' : '#FF4757';
      ctx.shadowBlur = isRevived ? 12 : 6;
      ctx.shadowColor = isRevived ? '#2DF89F' : '#FF4757';
      ctx.stroke();

      lastY = currentY;
      scanX += 3;
      if (scanX > width) {
        scanX = 0;
      }

      animFrameId = requestAnimationFrame(drawECG);
    }
    drawECG();
